function initFirebase(){
  if(FIREBASE_CONFIG.apiKey.includes("PASTE_YOUR")){
    setSyncStatus("local","Local only — Firebase not configured");
    try{const d=localStorage.getItem("effica_obs_v2");if(d)data=JSON.parse(d);}catch(e){}
    render(); return;
  }
  try{
    firebase.initializeApp(FIREBASE_CONFIG);
    db=firebase.database(); firebaseOk=true;
    setSyncStatus("connecting","Connecting…");
    db.ref(DB_PATH).on("value",snap=>{
      const v=snap.val();
      if(v){data=v;}
      else{db.ref(DB_PATH).set(data);}
      render();
      setSyncStatus("live","Live — synced to all users");
    },()=>{setSyncStatus("error","Firebase error");});
    db.ref(".info/connected").on("value",snap=>{
      setSyncStatus(snap.val()?"live":"connecting", snap.val()?"Live — synced to all users":"Reconnecting…");
    });
  }catch(e){setSyncStatus("error","Firebase error"); render();}
}
function setSyncStatus(s,m){
  const d=document.getElementById("sync-dot");
  d.className="sync-dot "+(s==="live"?"live":s==="saving"?"saving":"");
  document.getElementById("sync-txt").textContent=m;
}
function saveData(){
  if(firebaseOk&&db){
    setSyncStatus("saving","Saving…");
    db.ref(DB_PATH).set(data).then(()=>setSyncStatus("live","Live — synced")).catch(()=>{});
  } else {
    try{localStorage.setItem("effica_obs_v2",JSON.stringify(data));}catch(e){}
  }
  showToast("✅ Saved");
}

/* ── BUILD ROLE TREE ── */
function buildRoleTree(dept){
  const roles=dept.roles;
  if(!roles.length) return `<div style="font-size:11px;color:#9CA3AF;padding:8px 0;font-style:italic;">No roles yet</div>`;
  const byGroup={};
  roles.forEach(r=>{const g=r.group||"Team";if(!byGroup[g])byGroup[g]=[];byGroup[g].push(r);});
  const groupRank=g=>Math.min(...byGroup[g].map(r=>TYPE_CFG[r.type]?.order??9));
  const sortedGroups=Object.keys(byGroup).sort((a,b)=>groupRank(a)-groupRank(b));
  const soloGroups=sortedGroups.filter(g=>byGroup[g].every(r=>r.type==="tl"||r.type==="mgr")||byGroup[g].length===1);
  const leafGroups=sortedGroups.filter(g=>!soloGroups.includes(g));
  let html="";

  soloGroups.forEach(g=>{
    byGroup[g].forEach(r=>{
      html+=`<div class="v-line" style="height:14px;"></div>${roleNodeHTML(r,dept.id)}`;
    });
  });

  if(leafGroups.length>0){
    html+=`<div class="v-line" style="height:14px;"></div>`;
    html+=`<div class="role-level" style="position:relative;gap:0;align-items:flex-start;">`;
    if(leafGroups.length>1) html+=`<div class="h-bar" style="left:30%;right:30%;"></div>`;
    leafGroups.forEach(g=>{
      const sortedR=[...byGroup[g]].sort((a,b)=>(TYPE_CFG[a.type]?.order??9)-(TYPE_CFG[b.type]?.order??9));
      const typeRows=[];let cur=null;
      sortedR.forEach(r=>{if(!cur||cur.type!==r.type){cur={type:r.type,items:[r]};typeRows.push(cur);}else cur.items.push(r);});
      html+=`<div class="role-node" style="margin:0 5px;">`;
      html+=`<div class="v-line" style="height:14px;"></div>`;
      if(leafGroups.length>1) html+=`<div style="font-size:9px;font-weight:700;color:#9CA3AF;letter-spacing:.5px;margin-bottom:4px;background:#F3F4F6;border-radius:20px;padding:1px 8px;">${g}</div>`;
      typeRows.forEach(row=>{
        html+=`<div style="position:relative;display:flex;gap:4px;justify-content:center;align-items:flex-start;">`;
        if(row.items.length>1) html+=`<div class="h-bar" style="left:25%;right:25%;"></div>`;
        row.items.forEach(r=>{
          html+=`<div style="display:flex;flex-direction:column;align-items:center;">`;
          if(row.items.length>1) html+=`<div class="v-line" style="height:12px;"></div>`;
          html+=roleNodeHTML(r,dept.id);
          html+=`</div>`;
        });
        html+=`</div><div class="v-line" style="height:8px;"></div>`;
      });
      html+=`</div>`;
    });
    html+=`</div>`;
  }
  return html;
}

function roleNodeHTML(r,deptId){
  const s=TYPE_CFG[r.type]||TYPE_CFG.tr;
  return `<div class="role-box" onclick="openEditRole('${deptId}','${r.id}')"
    style="border-color:#E5E7EB;"
    onmouseover="this.style.background='${s.bg}';this.style.borderColor='${s.border}';"
    onmouseout="this.style.background='#fff';this.style.borderColor='#E5E7EB';">
    <div class="role-box-icon">${s.icon}</div>
    <div class="role-box-title">${r.role}</div>
    ${r.name?`<div class="role-box-name">${r.name}</div>`:""}
  </div>`;
}

function saveAddRole(deptId){
  const role=document.getElementById("ar_r").value.trim();
  if(!role){alert("Please enter a role title.");return;}
  const d=data.departments.find(x=>x.id===deptId);
  const grpNew=document.getElementById("ar_ng").value.trim();
  const grpSel=document.getElementById("ar_g").value;
  d.roles.push({id:uid(),role,name:document.getElementById("ar_n").value.trim(),type:document.getElementById("ar_t").value||"jr",group:grpNew||grpSel||"Team"});
  d.headcount++;
  saveData();render();closeOverlay();
}
